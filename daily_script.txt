name: USA Dose Daily Automation

on:
  workflow_dispatch:
  schedule:
    - cron: "30 13 * * *"

permissions:
  contents: write

jobs:
  generate:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Setup Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: Install Gemini SDK
        run: |
          python -m pip install --upgrade pip
          python -m pip install google-genai

      - name: Install Free TTS
        run: |
          sudo apt-get update
          sudo apt-get install -y espeak-ng

      - name: Generate AI USA Short Script
        env:
          GEMINI_API_KEY: ${{ secrets.GEMINI_API_KEY }}
        run: |
          python scripts/generate_script.py

      - name: Generate English Voice
        run: |
          python scripts/generate_voice.py

      - name: Check Generated Files
        run: |
          echo "Generated files:"
          ls -lh daily_script.txt voice.mp3

      - name: Save generated files
        run: |
          git config user.name "USA Dose Bot"
          git config user.email "actions@users.noreply.github.com"

          git add daily_script.txt voice.mp3

          if git diff --cached --quiet; then
            echo "No changes to commit."
          else
            git commit -m "Generate daily USA Short voice"
            git push
          fi

name: USA Dose Daily Automation

on:
  workflow_dispatch:
  schedule:
    - cron: "30 13 * * *"

permissions:
  contents: write

jobs:
  generate:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Setup Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: Install Gemini SDK
        run: pip install -U google-genai

      - name: Generate AI USA Short Script
        env:
          GEMINI_API_KEY: ${{ secrets.GEMINI_API_KEY }}
        run: python scripts/generate_script.py

      - name: Save generated script
        run: |
          git config user.name "USA Dose Bot"
          git config user.email "actions@users.noreply.github.com"
          git add daily_script.txt
          git diff --cached --quiet || git commit -m "Generate AI USA script"
          git push
